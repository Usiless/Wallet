final class DashboardScreen extends ConsumerWidget {
  const DashboardScreen({super.key});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final balances = ref.watch(accountBalancesProvider);
    final pendingCount =
        ref.watch(pendingCandidatesProvider).value?.length ?? 0;
    return Scaffold(
      floatingActionButton: const MovementFab(),
      body: SafeArea(
        child: AsyncValueView(
          value: balances,
          data: (items) {
            final netWorth = items
                .where((item) => item.includeInNetWorth)
                .fold<int>(
                  0,
                  (sum, item) =>
                      sum +
                      (item.type == AccountType.creditCard
                          ? -item.balanceMinor
                          : item.balanceMinor),
                );
            final available = items
                .where(
                  (item) =>
                      item.includeInAvailableMoney &&
                      item.type != AccountType.creditCard,
                )
                .fold<int>(0, (sum, item) => sum + item.balanceMinor);
            final payments = ref.watch(pendingPaymentsProvider);
            final open = <PendingPayment>[...payments.value ?? []]
              ..removeWhere((p) => p.cancelled || p.remainingMinor <= 0);
            open.sort((a, b) => a.dueDate.compareTo(b.dueDate));
            final now = DateTime.now();
            final monthPending = open
                .where(
                  (p) =>
                      p.dueDate.year == now.year &&
                      p.dueDate.month == now.month,
                )
                .fold<int>(0, (sum, p) => sum + p.remainingMinor);
            final totalPending = open.fold<int>(
              0,
              (sum, p) => sum + p.remainingMinor,
            );
            return ListView(
              padding: const EdgeInsets.fromLTRB(16, 16, 16, 100),
              children: [
                _HeroBalance(
                  available: available,
                  inbox: IconButton(
                    tooltip: pendingCount > 0
                        ? 'Movimientos pendientes ($pendingCount)'
                        : 'Movimientos pendientes',
                    onPressed: () => context.push('/inbox'),
                    color: Colors.white,
                    icon: const PendingCountBadge(
                      child: Icon(Icons.inbox_outlined),
                    ),
                  ),
                ),
                const SizedBox(height: 16),
                ListTile(
                  contentPadding: EdgeInsets.zero,
                  title: const Text('Pendiente del mes'),
                  subtitle: MoneyText(monthPending),
                  trailing: const Icon(Icons.chevron_right),
                  onTap: () => context.push('/payments'),
                ),
                ListTile(
                  key: const ValueKey('overview-net-worth'),
                  contentPadding: EdgeInsets.zero,
                  title: const Text('Patrimonio total'),
                  subtitle: MoneyText(netWorth - totalPending),
                ),
                const SizedBox(height: 24),
                Row(
                  children: [
                    Expanded(
                      child: Text(
                        'Próximos pagos',
                        style: Theme.of(context).textTheme.titleLarge,
                      ),
                    ),
                    TextButton(
                      onPressed: () => context.push('/payments'),
                      child: const Text('Ver todos'),
                    ),
                  ],
                ),
                if (payments.hasError)
                  const Text('No se pudieron cargar los pagos.'),
                if (payments.isLoading) const LinearProgressIndicator(),
                if (!payments.isLoading && !payments.hasError && open.isEmpty)
                  const Text('Sin pagos pendientes.'),
                for (final payment in open.take(2))
                  ListTile(
                    contentPadding: EdgeInsets.zero,

  Future<void> _edit(BuildContext context, AccountRecord account) async {
    try {
      final db = ref.read(databaseProvider);
      final aliases = await (db.select(
        db.accountAliases,
      )..where((t) => t.accountId.equals(account.id))).get();
      final card = await (db.select(
        db.creditCardProfiles,
      )..where((t) => t.accountId.equals(account.id))).getSingleOrNull();
      if (!context.mounted) return;
      final draft = await showCreateAccountDialog(
        context,
        initial: CreateAccountDraft(
          name: account.name,
          type: AccountType.values.byName(account.type),
          currencyCode: account.currencyCode,
          initialBalanceMinor: account.initialBalanceMinor,
          aliases: aliases.map((a) => a.alias).toList(),
          includeInAvailableMoney: account.includeInAvailableMoney,
          includeInNetWorth: account.includeInNetWorth,
          creditLimitMinor: card?.creditLimitMinor,
          closingDay: card?.closingDay,
          dueDay: card?.dueDay,
        ),
      );
      if (draft == null || !mounted) return;
      await ref.read(accountsServiceProvider).edit(account.id, draft);
    } catch (error) {
      if (context.mounted) {
        ScaffoldMessenger.of(context)
            .showSnackBar(SnackBar(content: Text('$error')));
      }
    }
  }

  Future<void> _archive(BuildContext context, AccountRecord account) async {
    try {
      await ref
          .read(accountsServiceProvider)
          .setArchived(account.id, !account.isArchived);
    } catch (error) {
      if (context.mounted) {
        ScaffoldMessenger.of(context)
            .showSnackBar(SnackBar(content: Text('$error')));
      }
    }
  }

  Future<void> _showCreate(BuildContext context, WidgetRef ref) async {
    final draft = await showCreateAccountDialog(context);
    if (draft == null || !context.mounted) return;
    try {
      await ref.read(accountsServiceProvider).createAccount(draft);
      if (context.mounted) {
        ScaffoldMessenger.of(context)
            .showSnackBar(const SnackBar(content: Text('Cuenta creada.')));
      }
    } catch (error) {
      if (context.mounted) {
        ScaffoldMessenger.of(context)
            .showSnackBar(SnackBar(content: Text('$error')));
      }
    }
  }
}

                      row: rows[index].transaction,
                      movement: rows[index],

                          showMovementDetail(context, ref, rows[index]),
                      onRefund: () => showRefundDialog(

    WidgetRef ref,

      await ledger.softDelete(row.id);

      if (messenger.mounted) {
        messenger.showSnackBar(SnackBar(content: Text('$error')));

              }
            }

      ),

final class _MovementTile extends StatelessWidget {
  const _MovementTile({
    required this.row,
    required this.onDelete,
    required this.onClone,
    required this.onEdit,
    required this.onRefund,
    required this.movement,
    required this.onDetail,
  });

  final VoidCallback onDelete;
  final VoidCallback onClone;
  final VoidCallback onEdit;
  final VoidCallback onRefund;

  Widget build(BuildContext context) {

        ? -movement.amountMinor
        : row.type == 'income' || row.type == 'refund'
        ? movement.amountMinor
        : 0;

                onRefund();

                onEdit();
              } else if (value == 'clone') {
                onClone();
              } else {
                onDelete();
              }
            },
            itemBuilder: (context) => [
              if (['income', 'expense', 'transfer'].contains(row.type))
                PopupMenuItem(
                  value: 'edit',
                  child: Text(movement.isQuota ? 'Editar compra' : 'Editar'),

                  value: 'refund',

    'income' => Icons.north_east,

    'refund' => Icons.replay,
    _ => Icons.tune,

    'expense' => 'Gasto',
    'income' => 'Ingreso',
    'transfer' => 'Transferencia',
    'refund' => 'Reembolso',
    _ => 'Ajuste',
  };

Future<void> showMovementDetail(
  BuildContext context,
  WidgetRef ref,
  MovementRow movement,
) async {
  final row = movement.transaction, db = ref.read(databaseProvider);
  final plans = await (db.select(
    db.installmentPlans,
  )..where((p) => p.transactionId.equals(row.id))).get();
  final entries = plans.isEmpty
      ? <InstallmentEntryRecord>[]
      : await (db.select(db.installmentEntries)
              ..where((e) => e.planId.equals(plans.single.id))
              ..orderBy([(e) => OrderingTerm.asc(e.installmentNumber)]))
            .get();
  final accountRowsFuture = db.select(db.accounts).get();
  final categoriesFuture = db.select(db.categories).get();
  final splitsFuture = (db.select(
    db.transactionSplits,
  )..where((split) => split.transactionId.equals(row.id))).get();
  final accountRows = await accountRowsFuture;
  final categories = await categoriesFuture;
  final splits = await splitsFuture;
  String accountName(String? id) =>
      accountRows.where((a) => a.id == id).firstOrNull?.name ?? '—';
  final categoryNames = splits
      .map(
        (split) =>
            categories
                .where((category) => category.id == split.categoryId)
                .firstOrNull
                ?.name ??
            'Categoría archivada',
      )
      .join(' / ');
  final channel = switch (row.paymentChannel) {
    'cash' => 'Efectivo',
    'physicalCard' => 'Tarjeta física',
    'debitCard' => 'Tarjeta de débito',
    'physicalCreditCard' => 'Tarjeta de crédito',
    'googleWalletContactless' => 'Google Wallet',
    'onlineCard' => 'Tarjeta online',
    'bankTransfer' => 'Transferencia bancaria',
    _ => 'Otro',
  };
  if (!context.mounted) return;
  await showModalBottomSheet<void>(
    context: context,
    showDragHandle: true,
    isScrollControlled: true,
    builder: (sheet) => SafeArea(
      child: FractionallySizedBox(
        heightFactor: .8,
        child: ListView(
          padding: const EdgeInsets.all(20),
          children: [
            Text(

            const SizedBox(height: 16),
            Text(row.description),
            MoneyText(
              movement.amountMinor,
              style: Theme.of(sheet).textTheme.headlineMedium,
            ),
            Text(
              '${MaterialLocalizations.of(sheet).formatShortDate(movement.occurredAt)}${movement.planned ? ' · Prevista' : ''}',
            ),
            const SizedBox(height: 24),
            _MovementFact(
              'Cuenta',
              accountName(row.fromAccountId ?? row.toAccountId),
            ),
            if (row.fromAccountId != null && row.toAccountId != null)
              _MovementFact('Destino', accountName(row.toAccountId)),
            if (splits.isNotEmpty) _MovementFact('Categoría', categoryNames),
            _MovementFact('Canal', channel),
            if (row.note.isNotEmpty) _MovementFact('Nota', row.note),
            if (entries.isNotEmpty) ...[
              const Divider(height: 32),
              ListTile(
                title: const Text('Compra original'),
                subtitle: Text(
                  MaterialLocalizations.of(sheet)
                      .formatShortDate(row.occurredAt.toLocal()),
                ),
                trailing: MoneyText(row.amountMinor),
              ),
              for (final entry in entries)
                ListTile(
                  title: Text(
                    'Cuota ${entry.installmentNumber}/${plans.single.installmentCount}',
                  ),
                  subtitle: Text(
                    MaterialLocalizations.of(sheet)
                        .formatShortDate(entry.periodStart.toUtc()),
                  ),
                  trailing: MoneyText(entry.amountMinor),
                ),
            ],
            if (['expense', 'income', 'transfer'].contains(row.type))
              FilledButton(
                onPressed: () {
                  Navigator.pop(sheet);
                  context.push('/transactions/new?edit=${row.id}');
                },
                child: Text(entries.isEmpty ? 'Editar' : 'Editar compra'),
              ),
            if (row.type == 'expense')
              OutlinedButton(
                onPressed: () {
                  Navigator.pop(sheet);
                  showRefundDialog(context, ref, row);
                },
                child: const Text('Registrar devolución'),
              ),
            if (['expense', 'income', 'transfer'].contains(row.type))
              TextButton(
                onPressed: () {
                  Navigator.pop(sheet);
                  context.push('/transactions/new?clone=${row.id}');
                },
                child: const Text('Clonar'),

            TextButton(
              onPressed: () async {
                final yes = await showDialog<bool>(
                  context: sheet,
                  builder: (dialog) => AlertDialog(
                    title: Text(
                      entries.isEmpty
                          ? 'Eliminar movimiento'
                          : 'Eliminar compra y sus cuotas',
                    ),
                    actions: [
                      TextButton(
                        onPressed: () => Navigator.pop(dialog, false),
                        child: const Text('Cancelar'),
                      ),
                      FilledButton(
                        onPressed: () => Navigator.pop(dialog, true),
                        child: const Text('Eliminar'),
                      ),
                    ],
                  ),
                );
                if (yes != true || !context.mounted) return;
                try {
                  final ledger = ref.read(ledgerServiceProvider);
                  await ledger.softDelete(row.id);
                  if (!sheet.mounted || !context.mounted) return;
                  Navigator.pop(sheet);
                  ScaffoldMessenger.of(context).showSnackBar(
                    SnackBar(
                      content: const Text('Movimiento eliminado.'),
                      action: SnackBarAction(
                        label: 'Deshacer',
                        onPressed: () async {
                          try {
                            await ledger.restore(row.id);

                            if (context.mounted) {
                              ScaffoldMessenger.of(context)

                  );
                } catch (e) {
                  if (sheet.mounted) {
                    ScaffoldMessenger.of(sheet)
                        .showSnackBar(SnackBar(content: Text('$e')));
                  }
                }
              },

/// Ordered arcs share one radius and expand their scale above the target.
final class BudgetRingSegment {
  const BudgetRingSegment(
    this.fraction, {
    required this.excess,
    required this.pending,
  });
  final double fraction;
  final bool excess, pending;
}

class BudgetRingPainter extends CustomPainter {
  const BudgetRingPainter(this.ratio, this.colors, {double? paidRatio})
    : paidRatio = paidRatio ?? ratio;
  final double ratio, paidRatio;
  final ColorScheme colors;

  List<BudgetRingSegment> get segments {
    final total = math.max(0.0, ratio), paid = paidRatio.clamp(0.0, total);
    final within = math.min(1.0, total), scale = math.max(1.0, total);
    return [
      BudgetRingSegment(
        math.min(paid, within) / scale,
        excess: false,
        pending: false,
      ),
      BudgetRingSegment(
        math.max(0.0, within - paid) / scale,
        excess: false,
        pending: true,
      ),
      BudgetRingSegment(
        math.max(0.0, paid - within) / scale,
        excess: true,
        pending: false,
      ),
      BudgetRingSegment(
        (total - math.max(paid, within)) / scale,
        excess: true,
        pending: true,
      ),
    ].where((segment) => segment.fraction > 0).toList();
  }

  @override
  void paint(Canvas canvas, Size size) {
    final center = size.center(Offset.zero),
        radius = math.min(size.width, size.height) / 2 - 5;
    final paint = Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth = 8;
    canvas.drawCircle(
      center,
      radius,
      paint..color = colors.surfaceContainerHighest,
    );
    var start = -math.pi / 2;
    for (final segment in segments) {
      final color = segment.excess ? colors.error : colors.primary;
      final sweep = 2 * math.pi * segment.fraction;
      canvas.drawArc(
        Rect.fromCircle(center: center, radius: radius),
        start,
        sweep,
        false,
        paint..color = color.withValues(alpha: segment.pending ? .42 : 1),
      );
      start += sweep;
    }
  }

  @override
  bool shouldRepaint(BudgetRingPainter old) =>
      old.ratio != ratio || old.paidRatio != paidRatio || old.colors != colors;
}

final class MoneyText extends StatelessWidget {
  const MoneyText(
    this.minorUnits, {
    this.currencyCode = 'PYG',
    this.style,
    this.signed = false,
    this.currencySymbol,
    this.maxLines = 1,
    super.key,
  });

  final String currencyCode;
  final TextStyle? style;
  final bool signed;
  final String? currencySymbol;
  final int? maxLines;

  @override
  Widget build(BuildContext context) {

      signed && minorUnits > 0 ? '+$formatted' : formatted,
      style: style,
      maxLines: maxLines,
      overflow: TextOverflow.ellipsis,
    );
  }

      final service = AppLockService(ref.read(settingsProvider));
      final changed = await service.setEnabled(value);
      if (changed && mounted) setState(() => _lock = value);
    } catch (error) {
      _message('$error');
    } finally {
      if (mounted) {
        setState(() {
          _busy = false;
          _progress = null;
        });
      }
    }
  }

  Future<void> _createBackup() async {
    final password = await _passwordDialog('Contraseña para el backup');
    if (password == null || !mounted) return;
    setState(() => _busy = true);
    try {
      final saved = await ref
          .read(backupServiceProvider)
          .createWithPicker(password);
      if (saved) _message('Backup cifrado guardado.');
    } catch (error) {
      _message('$error');
    } finally {
      if (mounted) {
        setState(() {
          _busy = false;
          _progress = null;
        });
      }
    }
  }

  Future<void> _restoreBackup() async {
    setState(() => _busy = true);
    try {
      final selected = await ref.read(backupServiceProvider).pickForRestore();
      if (selected == null || !mounted) return;
      final (bytes, summary) = selected;
      final accepted = await _confirmRestore(summary);
      if (!accepted || !mounted) return;
      final password = await _passwordDialog('Contraseña del backup');
      if (password == null) return;
      await ref
          .read(backupServiceProvider)
          .restore(
            bytes,
            password,
            onProgress: (value) {
              if (mounted) setState(() => _progress = value);
            },
          );
      if (!mounted) return;
      ref.invalidate(settingsProvider);
      _message('Backup restaurado.');
      context.go('/start');
    } catch (error) {
      _message('$error');
    } finally {
      if (mounted) {
        setState(() {
          _busy = false;
          _progress = null;
        });
      }
    }
  }

import 'package:finanzas_locales/app/providers.dart';
import 'package:finanzas_locales/core/database/app_database.dart';
import 'package:finanzas_locales/core/money/money.dart';
import 'package:finanzas_locales/features/accounts/data/accounts_service.dart';
import 'package:finanzas_locales/features/transactions/domain/ledger_models.dart';
import 'package:finanzas_locales/features/transactions/presentation/transactions_screen.dart';
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:flutter_test/flutter_test.dart';

void main() {
  testWidgets(
    'detalle conserva cuenta canal y nota como hechos del movimiento',
    (tester) async {
      final db = AppDatabase.forTesting(NativeDatabase.memory());
      final container = ProviderContainer(
        overrides: [databaseProvider.overrideWithValue(db)],
      );
      addTearDown(() async {
        container.dispose();
        await db.close();
      });
      final account = await container
          .read(accountsServiceProvider)
          .createAccount(
            const CreateAccountDraft(name: 'Caja', type: AccountType.cash),
          );
      await container
          .read(ledgerServiceProvider)
          .create(
            TransactionDraft(
              type: TransactionType.income,
              amount: const Money(10000),
              occurredAt: DateTime.now(),
              toAccountId: account,
              description: 'Cobro',
              note: 'Referencia 42',
              paymentChannel: PaymentChannel.cash,
            ),
          );
      await tester.pumpWidget(
        UncontrolledProviderScope(
          container: container,
          child: const MaterialApp(home: TransactionsScreen()),
        ),
      );
      await tester.pumpAndSettle();
      await tester.tap(find.text('Cobro'));
      await tester.pumpAndSettle();
      expect(find.text('Cuenta'), findsOneWidget);
      expect(find.text('Canal'), findsOneWidget);
      expect(find.text('Nota'), findsOneWidget);
      expect(find.text('Referencia 42'), findsOneWidget);
      expect(tester.takeException(), isNull);
      await tester.pumpWidget(const SizedBox());
      await tester.pumpAndSettle();
    },
  );
}
