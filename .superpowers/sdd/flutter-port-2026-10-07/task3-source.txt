  Future<void> _loadSavedTemplate(String id) async {
    _loadingTemplate = true;
    _templateError = null;
    try {
      final template = await ref.read(manualTemplateServiceProvider).read(id);
      if (template == null) throw StateError('La plantilla ya no existe.');
      if (!mounted) return;
      final f = template.fields;
      final merchantId = f['merchantId'] as String?;
      final linkedMerchant = merchantId == null
          ? null
          : await (ref
                    .read(databaseProvider)
                    .select(ref.read(databaseProvider).merchants)
                  ..where((t) => t.id.equals(merchantId)))
                .getSingleOrNull();
      if (!mounted) return;
      setState(() {
        _templateName = template.name;
        if (widget.editTemplate && f['schedule'] is Map) {
          _templateSchedule = Map<String, dynamic>.from(f['schedule'] as Map);
        }
        _type = TransactionType.values.byName(f['type'] as String);
        _amount.text = f['amountMinor'] == null
            ? ''
            : Money(f['amountMinor'] as int).format(withCode: false).trim();
        _from = f['from'] as String?;
        _to = f['to'] as String?;
        _category = f['category'] as String?;
        _selectedMerchant = linkedMerchant?.isArchived == false
            ? linkedMerchant
            : null;
        _merchant.text =
            _selectedMerchant?.name ?? f['merchant'] as String? ?? '';
        _note.text = f['note'] as String? ?? '';
        _channel = PaymentChannel.values.byName(
          f['channel'] as String? ?? 'other',
        );
        _installments = f['installments'] as bool? ?? false;
        _installmentCount = f['installmentCount'] as int? ?? 2;
        _reminders = f['reminders'] is Map
            ? PaymentReminderSettings.fromJson(f['reminders'] as Map)
            : const PaymentReminderSettings();
        _splitEnabled = f['splitEnabled'] as bool? ?? false;
        for (final item in f['splits'] as List? ?? []) {
          _splits.add(
            _SplitInput(categoryId: item['category'] as String?)
              ..amount.text = item['amountMinor'] == null
                  ? ''
                  : Money(item['amountMinor'] as int)
                        .format(withCode: false)
                        .trim(),
          );
        }
        _advanced = _installments || _note.text.isNotEmpty;
        if (widget.savedTemplateId != null) _baseline = _draftSnapshot;
      });
    } catch (error) {
      if (mounted) _templateError = '$error';
    } finally {
      if (mounted) setState(() => _loadingTemplate = false);
    }
  }

    DateTime nextDate = _templateSchedule?['nextDate'] is String
        ? parseScheduledDate(_templateSchedule!['nextDate'] as String)
        : afterRegistration
        ? nextScheduledDate(_occurredAt, {
            'frequency': 'monthly',
            'anchorDay': _occurredAt.day,
          })
        : DateTime.now();
    bool choseDate = false;
    final originalDate = scheduledDate(nextDate);
    final originalFrequency = frequency;
    final accepted = await showDialog<bool>(
      context: context,
      builder: (context) => StatefulBuilder(
        builder: (context, update) => AlertDialog(
          title: const Text('Guardar plantilla'),
          content: SingleChildScrollView(

              mainAxisSize: MainAxisSize.min,
              children: [

                ),

                          !choseDate) {

                          'anchorDay': _occurredAt.day,

                      if (chosen != null && context.mounted) {

                          nextDate = chosen;

                      }
                    },

      }

  Future<void> _loadClone(String id) async {
    final database = ref.read(databaseProvider);
    setState(() => _loadingTemplate = true);
    try {
      final row = await (database.select(
        database.ledgerTransactions,
      )..where((t) => t.id.equals(id))).getSingleOrNull();
      if (row == null || row.status != 'confirmed') {
        throw StateError('El movimiento ya no está disponible.');
      }
      final splits = await (database.select(
        database.transactionSplits,
      )..where((t) => t.transactionId.equals(id))).get();
      final plan = await (database.select(
        database.installmentPlans,
      )..where((t) => t.transactionId.equals(id))).getSingleOrNull();
      final merchant = row.merchantId == null
          ? null
          : await (database.select(
              database.merchants,
            )..where((t) => t.id.equals(row.merchantId!))).getSingleOrNull();
      final oldAccounts =
          await (database.select(database.accounts)..where(
                (t) => t.id.isIn(
                  [row.fromAccountId, row.toAccountId].whereType<String>(),
                ),
              ))
              .get();
      final oldCategories = await (database.select(
        database.categories,
      )..where((t) => t.id.isIn(splits.map((s) => s.categoryId)))).get();
      if (!mounted) return;
      setState(() {
        _editingAccounts = oldAccounts;
        _editingCategories = oldCategories;
        _type = TransactionType.values.byName(row.type);
        _amount.text = Money(row.amountMinor).format(withCode: false).trim();
        _originalMerchantId = widget.editId != null ? row.merchantId : null;
        _selectedMerchant =
            widget.editId != null || merchant?.isArchived == false
            ? merchant
            : null;
        _merchant.text = _selectedMerchant?.name ?? row.description;
        _note.text = row.note;
        _from = row.fromAccountId;
        _to = row.toAccountId;
        _channel = PaymentChannel.values.byName(row.paymentChannel);
        _occurredAt = widget.editId != null
            ? row.occurredAt.toLocal()
            : DateTime.now();
        _editingUpdatedAt = row.updatedAt;
        _installments = plan != null;
        _installmentCount = plan?.installmentCount ?? 2;
        _firstInstallmentDate = widget.editId != null
            ? plan?.firstPeriod.toUtc()
            : null;
        _splitEnabled = splits.length > 1;
        _category = splits.length == 1 ? splits.single.categoryId : null;
        for (final split in splits.where((_) => _splitEnabled)) {
          _splits.add(
            _SplitInput(categoryId: split.categoryId)
              ..amount.text = Money(split.amountMinor)
                  .format(withCode: false)
                  .trim(),
          );
        }
        _advanced = true;
        _baseline = _draftSnapshot;
      });
    } catch (error) {
      if (mounted) _templateError = '$error';
    } finally {
      if (mounted) setState(() => _loadingTemplate = false);
    }
  }

final class FinanzasApp extends ConsumerStatefulWidget {
  const FinanzasApp({super.key});

  @override
  ConsumerState<FinanzasApp> createState() => _FinanzasAppState();
}

class _FinanzasAppState extends ConsumerState<FinanzasApp>
    with WidgetsBindingObserver {
  bool _processing = false;
  bool _initialReminderActionRead = false;
  bool _syncingReminders = false;
  bool _reminderRefreshRequested = false;
  bool _refreshRequested = false;
  final _messenger = GlobalKey<ScaffoldMessengerState>();
  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    ref.read(nativeNotificationBridgeProvider).setCaptureHandler(() async {
      if (mounted &&
          WidgetsBinding.instance.lifecycleState == AppLifecycleState.resumed) {
        await _ingestNativeQueue();
      }
    });
    WidgetsBinding.instance.addPostFrameCallback((_) => _ingestNativeQueue());
  }

  Future<void> _reconcileReminders() async {
    if (!_initialReminderActionRead || !mounted) return;
    if (_syncingReminders) {
      _reminderRefreshRequested = true;
      return;
    }
    _syncingReminders = true;
    try {
      do {
        _reminderRefreshRequested = false;
        final payments = await ref.read(pendingPaymentServiceProvider).list();
        if (!mounted) return;
        await synchronizePaymentReminders(
          ref.read(paymentReminderBridgeProvider),
          payments,
        );
      } while (_reminderRefreshRequested && mounted);
    } catch (_) {
      // Storage and platform failures are retried on the next change/resume;
      // permission is requested only from the explicit reminder controls.
    } finally {
      _syncingReminders = false;
    }
  }

  Future<void> _ingestNativeQueue() async {
    if (_processing) {
      _refreshRequested = true;
      return;
    }
    _processing = true;
    try {
      try {
        final paymentId = await ref
            .read(paymentReminderBridgeProvider)
            .consumeOpenAction();
        if (mounted && paymentId != null) {
          final payment = await ref
              .read(pendingPaymentServiceProvider)
              .read(paymentId);
          if (mounted)
            ref
                .read(routerProvider)
                .go(
                  payment == null
                      ? '/payments'
                      : '/payments/${Uri.encodeComponent(paymentId)}',
                );
        }
      } catch (_) {
        // Platform channel is unavailable in host rendering/tests.
      } finally {
        _initialReminderActionRead = true;
      }
      final action = await ref
          .read(nativeNotificationBridgeProvider)
          .getInitialAction();
      if (action == 'py.finanzaslocales.QUICK_EXPENSE') {
        ref.read(routerProvider).go('/transactions/new?type=expense');
      } else if (action == 'py.finanzaslocales.OPEN_INBOX') {
        ref.read(routerProvider).go('/inbox');
      }
      final settings = ref.read(settingsProvider);
      final wallet = await settings.readBool('notification_listener_enabled');
      final sms =
          developerToolsEnabled &&
          await settings.readBool('sms_notifications_enabled');
      if (!developerToolsEnabled) {
        await settings.write('diagnostic_mode', 'false');
        await settings.write('sms_notifications_enabled', 'false');
      }
      await ref
          .read(nativeNotificationBridgeProvider)
          .configure(
            enabled: wallet || sms,
            diagnosticMode:
                developerToolsEnabled &&
                await settings.readBool('diagnostic_mode'),
            allowedPackages: {
              if (wallet) ...{
                'com.google.android.gms',
                'com.google.android.apps.walletnfcrel',
              },
              if (sms) 'com.google.android.apps.messaging',
            },
          );
      final events = await ref
          .read(nativeNotificationBridgeProvider)
          .queuedEnvelopes();
      for (final event in events) {
        if (event.isTest && !developerToolsEnabled) continue;
        if (event.packageName == 'com.google.android.apps.messaging' && sms) {
          await ref.read(smsNotificationIngestionProvider).ingest(event);
        } else if (wallet || (developerToolsEnabled && event.isTest)) {
          await ref.read(notificationIngestionProvider).process(event);
        }
        await ref.read(nativeNotificationBridgeProvider).acknowledge(event);
      }
      await ref.read(notificationIngestionProvider).purgeExpiredRawEvents();
    } catch (_) {
      // Android access may be unavailable; explicit refresh remains available.
    } finally {
      try {
        final result = await ref
            .read(scheduledTemplateServiceProvider)
            .processDue();
        if (mounted &&
            (result.registered.isNotEmpty || result.pending.isNotEmpty)) {
          _messenger.currentState?.showSnackBar(
            SnackBar(
              content: Text(
                '${result.registered.length} pagos pendientes preparados. ${result.pending.length} por revisar.',
              ),
              action: SnackBarAction(
                label: 'Ver',
                onPressed: () => ref.read(routerProvider).push('/payments'),
              ),
            ),
          );
        }
      } catch (_) {
        // Failed storage access leaves the schedule intact for the next opening.
      }
      try {
        await ref.read(budgetServiceProvider).ensureCurrentMonth();
      } catch (_) {
        if (mounted) {
          _messenger.currentState?.showSnackBar(
            const SnackBar(
              content: Text(
                'No se pudieron preparar los presupuestos mensuales. Volvé a intentar desde Análisis.',
              ),
            ),
          );
        }
      }
      await _reconcileReminders();
      _processing = false;
      if (_refreshRequested && mounted) {
        _refreshRequested = false;
        await _ingestNativeQueue();
      }
    }
  }

  @override
  void dispose() {
    ref.read(nativeNotificationBridgeProvider).setCaptureHandler(null);
    WidgetsBinding.instance.removeObserver(this);
    super.dispose();
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.resumed) _ingestNativeQueue();
  }

  @override
  Widget build(BuildContext context) {
    ref.listen(pendingPaymentsProvider, (_, next) {
      if (next.hasValue) _reconcileReminders();
    });
    final router = ref.watch(routerProvider);
    final lock = AppLockService(ref.watch(settingsProvider));
    return MaterialApp.router(
      scaffoldMessengerKey: _messenger,
      title: 'Cabal',
      debugShowCheckedModeBanner: false,
      theme: AppTheme.light(),
      darkTheme: AppTheme.dark(),
      themeMode: ThemeMode.system,
      locale: const Locale('es', 'PY'),
      supportedLocales: const [Locale('es', 'PY'), Locale('es')],
      localizationsDelegates: const [
        GlobalMaterialLocalizations.delegate,
        GlobalWidgetsLocalizations.delegate,
        GlobalCupertinoLocalizations.delegate,
      ],
      routerConfig: router,
      builder: (context, child) =>
          AppLockGate(service: lock, child: child ?? const SizedBox.shrink()),
    );
  }
}
